// ========================================== // CÓDIGO COMPLETO PARA TU FORMULARIO PÚBLICO WEB // ========================================== const API_URL = "https://script.google.com/macros/s/AKfycbw3L8PHJ4teJL1aOvpGNdFBnz_STLvwXjuLQaAV2tFl-Jy1tVI-RbRc5th2SbP7eAqQog/exec"; let dbUbigeo = []; let dbClientes = []; let dbShalom = []; // 1. Funciones Auxiliares de Normalización function normalizarTexto(txt) { if (!txt) return ""; return String(txt) .toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .trim(); } function normalizarTelefono(num) { if (!num) return ""; const soloDigitos = String(num).replace(/\D/g, ""); return soloDigitos.length >= 9 ? soloDigitos.slice(-9) : soloDigitos; } // 2. Extraer ÚNICAMENTE el celular limpio desde el Token de la URL function obtenerCelularDesdeToken() { const urlParams = new URLSearchParams(window.location.search); const token = urlParams.get('token'); if (!token) return ""; try { const rawData = atob(token); // Ejemplo: "903317718_1786339297140" const celularLimpio = rawData.split('_')[0].replace(/\D/g, ""); // Extrae solo "903317718" return celularLimpio.length >= 9 ? celularLimpio.slice(-9) : celularLimpio; } catch (e) { console.error("Error al decodificar token:", e); return ""; } } // 3. Cargar la base de datos completa de Google Sheets async function cargarBaseDatosWeb() { try { const res = await fetch(API_URL, { method: "GET", redirect: "follow" }); const data = await res.json(); dbUbigeo = data.ubigeo || []; dbClientes = data.clientes || []; dbShalom = data.shalom || []; poblarDepartamentosWeb(); } catch (err) { console.error("Error al conectar con Google Sheets:", err); } } // 4. Poblar Listas Desplegables de Ubigeo function poblarDepartamentosWeb() { const selDep = document.getElementById("crm-dep") || document.getElementById("departamento"); if (!selDep || dbUbigeo.length === 0) return; const departamentos = [...new Set( dbUbigeo .map(item => String(item.departamento || item.DEPARTAMENTO || "").trim()) .filter(dep => dep !== "" && !dep.toUpperCase().includes("DEPARTAMEN")) )].sort(); let html = ''; departamentos.forEach(dep => { html += ``; }); selDep.innerHTML = html; } function poblarProvinciasWeb(depSeleccionado) { const selProv = document.getElementById("crm-prov") || document.getElementById("provincia"); const selDist = document.getElementById("crm-dist") || document.getElementById("distrito"); if (!selProv || !selDist) return; selProv.innerHTML = ''; selDist.innerHTML = ''; if (!depSeleccionado) return; const provincias = [...new Set( dbUbigeo .filter(item => normalizarTexto(item.departamento || item.DEPARTAMENTO) === normalizarTexto(depSeleccionado)) .map(item => String(item.provincia || item.PROVINCIA || "").trim()) .filter(prov => prov !== "" && prov.toUpperCase() !== "PROVINCIA") )].sort(); let html = ''; provincias.forEach(prov => { html += ``; }); selProv.innerHTML = html; } function poblarDistritosWeb(depSeleccionado, provSeleccionada) { const selDist = document.getElementById("crm-dist") || document.getElementById("distrito"); if (!selDist) return; selDist.innerHTML = ''; if (!depSeleccionado || !provSeleccionada) return; const distritos = [...new Set( dbUbigeo .filter(item => { const d = item.departamento || item.DEPARTAMENTO || ""; const p = item.provincia || item.PROVINCIA || ""; return normalizarTexto(d) === normalizarTexto(depSeleccionado) && normalizarTexto(p) === normalizarTexto(provSeleccionada); }) .map(item => String(item.distrito || item.DISTRITO || "").trim()) .filter(dist => dist !== "" && dist.toUpperCase() !== "DISTRITO") )].sort(); let html = ''; distritos.forEach(dist => { html += ``; }); selDist.innerHTML = html; } // 5. Cargar y Autocompletar Información del Cliente Existente function autocompletarDatosCliente(celular) { if (!celular || dbClientes.length === 0) return; const celBuscado = normalizarTelefono(celular); const clienteEncontrado = dbClientes.find(c => { let telSheet = ""; Object.keys(c).forEach(k => { if (k.toLowerCase().includes("tel") || k.toLowerCase().includes("cel") || k.toLowerCase().includes("num")) { telSheet = c[k]; } }); return normalizarTelefono(telSheet) === celBuscado; }); if (clienteEncontrado) { Object.keys(clienteEncontrado).forEach(k => { const key = normalizarTexto(k); const val = clienteEncontrado[k]; if (key.includes("dni") || key.includes("carnet")) { const inputDni = document.getElementById("crm-dni") || document.getElementById("dni"); if (inputDni && val) inputDni.value = val; } if (key.includes("nom") || key.includes("apell")) { const inputNom = document.getElementById("crm-nombre") || document.getElementById("nombre"); if (inputNom && val) inputNom.value = val; } if (key.includes("dep")) { const selDep = document.getElementById("crm-dep") || document.getElementById("departamento"); if (selDep && val) { selDep.value = val; poblarProvinciasWeb(val); } } if (key.includes("prov")) { const selProv = document.getElementById("crm-prov") || document.getElementById("provincia"); if (selProv && val) { selProv.value = val; const selDep = document.getElementById("crm-dep") || document.getElementById("departamento"); poblarDistritosWeb(selDep ? selDep.value : "", val); } } if (key.includes("dist")) { const selDist = document.getElementById("crm-dist") || document.getElementById("distrito"); if (selDist && val) selDist.value = val; } if (key.includes("dir") || key.includes("sede")) { const inputSede = document.getElementById("crm-direccion-sede") || document.getElementById("direccion"); if (inputSede && val) inputSede.value = val; } }); } } // 6. Escuchar Cambios en los Desplegables de Ubigeo function configurarEventosUbigeoWeb() { const selDep = document.getElementById("crm-dep") || document.getElementById("departamento"); const selProv = document.getElementById("crm-prov") || document.getElementById("provincia"); if (selDep) { selDep.addEventListener("change", () => { poblarProvinciasWeb(selDep.value); }); } if (selProv) { selProv.addEventListener("change", () => { poblarDistritosWeb(selDep ? selDep.value : "", selProv.value); }); } } // 7. Inicializar Formulario al Cargar la Página document.addEventListener("DOMContentLoaded", async () => { const celularLimpio = obtenerCelularDesdeToken(); const inputCelular = document.getElementById("crm-telefono") || document.getElementById("celular"); // Asignar el número de celular limpio (sin _timestamp) if (inputCelular && celularLimpio) { inputCelular.value = celularLimpio; } // Cargar BD y llenar campos existentes await cargarBaseDatosWeb(); configurarEventosUbigeoWeb(); if (celularLimpio) { autocompletarDatosCliente(celularLimpio); } });
